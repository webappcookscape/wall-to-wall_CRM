  {
    if(arguments[0] != null)
    {
      swal({

          cancelButtonText:  "Cancel",
          closeOnConfirm: false,
          closeOnCancel: false
       },
       function(isConfirm)
       {
         if (isConfirm)
         {
            //swal("", "", "success");
            location.href = url;

          } else 
          {
            swal("This operation has been cancelled", "", "error");
            //e.preventDefault();
          }
       });
     
    }
    else
    {
      return false;
    }
    return;
  }
